import Stripe from "stripe"; const stripe = new Stripe(process.env.STRIPE_SECRET_KEY); export default async function handler(req, res) { if (req.method !== "POST") { return res.status(405).json({ error: "Method not allowed" }); } try { const session = await stripe.checkout.sessions.create({ mode: "payment", line_items: req.body.items, success_url: "https://YOUR-NEW-VERCEL-URL.vercel.app/success", cancel_url: "https://YOUR-NEW-VERCEL-URL.vercel.app/cancel", }); res.status(200).json({ url: session.url }); } catch (err) { res.status(500).json({ error: err.message }); } }
